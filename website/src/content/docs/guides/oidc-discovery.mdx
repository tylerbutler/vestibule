import CodeBlock from "../../../components/CodeBlock.astro";

<section id="discover">
  <h2>Discover a provider</h2>
  <p>
    <code>vestibule_oidc.discover</code> builds a Vestibule strategy for a
    standards-compliant OpenID Connect provider. Supply the provider's issuer
    URL. The function reads the provider's{" "}
    <code>/.well-known/openid-configuration</code> document and returns the
    strategy.
  </p>
  <CodeBlock code={discoverCode} lang="gleam" />
  <p>
    Use the discovered strategy with the same two-phase flow and registry. You
    can also use it with
    <a href="/docs/packages/wisp">vestibule_wisp</a> or
    <a href="/docs/packages/mist">vestibule_mist</a> middleware.
  </p>
</section>

<section id="self-hosted">
  <h2>Self-hosted providers</h2>
  <p>
    <code>vestibule_oidc.discover</code> also supports self-hosted providers
    such as <a href="https://pocket-id.org">Pocket ID</a>. Supply the base URL of
    your instance. Then use <code>config.new</code> to add your client
    credentials to the discovered strategy.
  </p>
  <CodeBlock code={selfHostedCode} lang="gleam" />
</section>

<section id="register-client">
  <h2>Register a client</h2>
  <p>
    Register a client with your provider before you start the authorization
    flow:
  </p>
  <ul className="check-list">
    <li>
      <strong>Redirect URI:</strong> Use the exact URI that you pass to{" "}
      <code>config.new</code>. For example, use{" "}
      <code>https://app.example.com/auth/oidc/callback</code>. Production
      redirect URIs and OIDC issuers must use HTTPS.
      <code>http://localhost</code> and <code>http://127.0.0.1</code> are
      permitted for local development only.
    </li>
    <li>
      <strong>Scopes:</strong> Request <code>openid email profile</code> so that
      Vestibule can return the user's ID, email address, and name.
      <code>user_info.email</code> returns a value only if the provider reports
      <code>email_verified</code>.
    </li>
    <li>
      <strong>Client credentials:</strong> Copy the client ID and secret into{" "}
      <code>config.new</code>.
    </li>
  </ul>
</section>

<section id="nonce" className="security-callout">
  <div>
    <h2>Nonce validation</h2>
    <p>
      A discovered strategy uses an OIDC <code>nonce</code>. Vestibule generates
      the nonce and adds it to the authorization request. It validates the nonce
      in the returned <code>id_token</code> during callback handling. Wisp and
      Mist middleware store the nonce for you. If you use core directly, store{" "}
      <code>authorization_request.nonce(auth_request)</code> and supply it as{" "}
      <code>expected_nonce</code> during callback handling.
    </p>
  </div>
</section>
