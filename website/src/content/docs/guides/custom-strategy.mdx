import CodeBlock from "../../../components/CodeBlock.astro";
import {
  authorizeUrl,
  exchangeAndFetch,
  packageToml,
  strategyType,
  strategyValue,
} from "../../../data/doc-snippets";

<section id="core-boundary" className="security-callout">
  <div>
    <h2>What the core handles</h2>
    <p>
      Vestibule generates and validates CSRF state. It manages PKCE and resolves
      configured scopes. It also adds PKCE parameters and sends token-refresh
      operations to the strategy.
    </p>
  </div>
  <ul className="check-list">
    <li>Your strategy builds provider URLs.</li>
    <li>Your strategy exchanges codes for credentials.</li>
    <li>Your strategy fetches and normalizes user info.</li>
  </ul>
</section>

<section id="strategy-type">
  <h2>The Strategy type</h2>
  <p>
    The provider name, default scopes, and builder functions define the
    contract. Vestibule carries the custom error payload type through{" "}
    <code>AuthError(e)</code>.
  </p>
  <CodeBlock code={strategyType} lang="gleam" />
</section>

<section id="authoring-steps" className="step-list">
  <div className="step-item">
    <span className="step-number">1</span>
    <div>
      <h2>Create the provider package</h2>
      <p>
        Put a provider in a separate package. This example creates a Twitch
        strategy package. It uses Vestibule and the OAuth helpers that the
        built-in strategies use.
      </p>
      <CodeBlock code={packageToml} lang="toml" />
    </div>
  </div>

  <div className="step-item">
    <span className="step-number">2</span>
    <div>
      <h2>Build the authorization URL</h2>
      <p>
        The function receives the resolved scopes and the state value. Return
        the provider URL. Vestibule adds the PKCE challenge.
      </p>
      <CodeBlock code={authorizeUrl} lang="gleam" />
    </div>
  </div>

  <div className="step-item">
    <span className="step-number">3</span>
    <div>
      <h2>Exchange the code and fetch the user</h2>
      <p>
        Send the authorization code to the provider's token endpoint. Parse the
        response into <code>ExchangeResult</code>. Then use the credentials and
        artifacts to get a stable provider UID and normalized{" "}
        <code>UserInfo</code>.
      </p>
      <CodeBlock code={exchangeAndFetch} lang="gleam" />
    </div>
  </div>

  <div className="step-item">
    <span className="step-number">4</span>
    <div>
      <h2>Expose the final strategy value</h2>
      <p>
        Export a <code>strategy()</code> function. Return{" "}
        <code>strategy.Strategy(e)</code> if the provider uses only built-in
        error kinds. Return <code>strategy.Strategy(YourError)</code> if the
        provider wraps a domain error payload with <code>error.custom</code>.
      </p>
      <CodeBlock code={strategyValue} lang="gleam" />
    </div>
  </div>
</section>

<section id="authoring-checklist">
  <h2>Authoring checklist</h2>
  <ul className="check-list">
    <li>Use provider_support helpers. Do not copy built-in internals unless necessary.</li>
    <li>Keep scopes minimal and provider-owned.</li>
    <li>Keep optional provider fields optional when you normalize user info.</li>
    <li>Document refresh-token behavior. Providers use different rotation rules and returned scopes.</li>
    <li>Add tests for URL building, token parsing, profile normalization, and failure responses.</li>
  </ul>
</section>
